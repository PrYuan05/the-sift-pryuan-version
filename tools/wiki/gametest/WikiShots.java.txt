    private static final String[] MOBS = {"bloombud", "blub", "dartback", "echo_golem", "fusefly", "harmonizer", "hunter",
            "hurler", "lootkin", "monarch", "mourner", "nester", "nuzzle", "pollinator", "prickle", "roamroot", "scavenger",
            "sculk_monstrosity", "seedling", "sentinel", "sift_warden", "singer", "slabber", "sprout", "stalker", "trapper",
            "tuner", "wobble"};
    private static final String[] CLIMATE = {"nuzzle", "lootkin", "slabber", "prickle", "dartback", "fusefly", "wobble",
            "nester", "pollinator", "seedling", "sprout", "sentinel"};
    private static final String[] SURFACE = {"willow_grove", "amber_dunes", "violet_slate_abyss", "crimson_strata_canyon",
            "lullaby_hills", "soul_caldera"};
    private static final String[] UNDERGROUND = {"withered_hollows", "red_zone_caves", "resonant_vaults", "spirit_deeps"};
    private static final String[] STRUCTURES = {"willow_hut", "dune_tomb", "humbler_husk", "strata_shrine", "abyss_obelisk",
            "echo_den"};
    private static final String[][] ARENAS = {{"monarch", "amber_dunes"}, {"harmonizer", "willow_grove"},
            {"sculk_monstrosity", "violet_slate_abyss"}};

    private static ResourceKey<net.minecraft.world.level.Level> sift() {
        return ResourceKey.create(Registries.DIMENSION, PySift.id("the_sift"));
    }

    private static void shot(ClientGameTestContext c, String name) {
        c.takeScreenshot(TestScreenshotOptions.of(name).disableCounterPrefix().withSize(1280, 800));
    }

    @Override
    public void runTest(ClientGameTestContext context) {
        context.runOnClient(mc -> {
            mc.options.renderDistance().set(12);
        });
        try (TestSingleplayerContext world = context.worldBuilder().setUseConsistentSettings(true).adjustSettings(s -> s.setGenerateStructures(true)).create()) {
            TestServerContext server = world.getServer();
            server.runCommand("gamemode spectator @p");
            server.runCommand("weather clear 1000000");
            context.waitTicks(20);
            context.getInput().pressKey(o -> o.keyToggleGui);
            context.waitTicks(2);
            String only = System.getenv().getOrDefault("WIKI_ONLY", "");
            try {
                StringBuilder sb = new StringBuilder();
                for (var key : BuiltInRegistries.ITEM.keySet()) {
                    if (key.getNamespace().equals(PySift.MOD_ID)) sb.append(key).append(System.lineSeparator());
                }
                java.nio.file.Files.writeString(java.nio.file.Path.of(System.getProperty("user.dir")).resolve("wiki_registry.txt"), sb.toString());
                System.out.println("WIKISHOT registry written");
                StringBuilder tabs = new StringBuilder();
                context.runOnClient(mc -> {
                    net.minecraft.world.item.CreativeModeTabs.tryRebuildTabContents(mc.player.connection.enabledFeatures(), true,
                            mc.level.registryAccess());
                    for (var e : BuiltInRegistries.CREATIVE_MODE_TAB.entrySet()) {
                        if (!e.getKey().identifier().getNamespace().equals(PySift.MOD_ID)) continue;
                        for (var st : e.getValue().getDisplayItems()) {
                            tabs.append(e.getKey().identifier().getPath()).append(' ')
                                    .append(BuiltInRegistries.ITEM.getKey(st.getItem())).append(System.lineSeparator());
                        }
                    }
                });
                java.nio.file.Files.writeString(java.nio.file.Path.of(System.getProperty("user.dir")).resolve("wiki_tabs.txt"), tabs.toString());
            } catch (Exception e) {
                System.out.println("WIKISHOT registry failed " + e);
            }
            if (only.isEmpty() || only.contains("mobs")) mobs(context, server);
            siftPlaces(context, server, only);
        }
    }

    // ------------------------------------------------------------------------------------------------------- mobs
    private void mobs(ClientGameTestContext context, TestServerContext server) {
        server.runCommand("fill -14 199 -14 14 199 14 minecraft:light_gray_concrete");
        server.runCommand("fill -14 200 -14 14 230 14 minecraft:air");
        List<String[]> specs = new ArrayList<>();
        for (String m : MOBS) {
            specs.add(new String[]{m, m, ""});
        }
        for (String m : CLIMATE) {
            specs.add(new String[]{m, m + "_tropical", ",\"fabric:attachments\":{\"pysift:climate_variant\":1}"});
            specs.add(new String[]{m, m + "_cold", ",\"fabric:attachments\":{\"pysift:climate_variant\":2}"});
        }
        String[] blub = {"soul_engorged", "carapace", "meadows", "ravines", "fungal"};
        for (int i = 0; i < blub.length; i++) {
            specs.add(new String[]{"blub", "blub_" + blub[i], ",Variant:" + i});
        }
        String filter = System.getenv().getOrDefault("WIKI_MOBS", "");
        java.util.Map<String, Double> scale = java.util.Map.of("harmonizer", 2.6, "monarch", 2.6, "sculk_monstrosity", 2.0,
                "dartback", 1.0, "sentinel", 1.0);
        for (String[] s : specs) {
            if (!filter.isEmpty() && !java.util.Arrays.asList(filter.split(",")).contains(s[1])) continue;
            try {
                EntityType<?> type = BuiltInRegistries.ENTITY_TYPE.getValue(PySift.id(s[0]));
                double k = scale.getOrDefault(s[0], 1.0);
                float h = (float) (type.getDimensions().height() * k), w = (float) (type.getDimensions().width() * k);
                double size = Math.max(h, w * 1.3);
                double dist = size * 1.7 + 2.2;
                double ty = 200 + h * 0.5;
                double cx = 0.5 + dist * 0.5, cz = 0.5 - dist * 0.866, cy = ty + dist * 0.32;
                float yaw = (float) Math.toDegrees(-Math.atan2(0.5 - cx, 0.5 - cz));
                float pitch = (float) Math.toDegrees(Math.atan2(cy - ty, dist));
                server.runCommand("tp @e[type=!player] 0 -500 0");
                server.runCommand("kill @e[type=item]");
                server.runCommand("time set noon");
                server.runCommand(String.format(Locale.ROOT, "tp @p %.2f %.2f %.2f %.1f %.1f", cx, cy - 1.62, cz, yaw, pitch));
                context.waitTicks(12);
                shot(context, "mob_" + s[1] + "_bg");
                server.runCommand(String.format(Locale.ROOT,
                        "summon pysift:%s 0.5 200 0.5 {NoAI:1b,Silent:1b,PersistenceRequired:1b,Rotation:[180f,0f]%s}", s[0], s[2]));
                context.waitTicks(30);
                shot(context, "mob_" + s[1]);
            } catch (Throwable t) {
                System.out.println("WIKISHOT failed mob " + s[1] + ": " + t);
            }
        }
        server.runCommand("tp @e[type=!player] 0 -500 0");
    }

    // ------------------------------------------------------------------------------------------- Sift biomes etc.
    private static String tpCommand(double x, double y, double z, double lx, double ly, double lz) {
        double dx = lx - x, dy = ly - y, dz = lz - z;
        float yaw = (float) Math.toDegrees(-Math.atan2(dx, dz));
        float pitch = (float) Math.toDegrees(-Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
        return String.format(Locale.ROOT, "execute in pysift:the_sift run tp @p %.1f %.1f %.1f %.1f %.1f", x, y - 1.62, z, yaw, pitch);
    }

    private static BlockPos findBiome(MinecraftServer s, String biome) {
        ServerLevel level = s.getLevel(sift());
        ResourceKey<Biome> key = ResourceKey.create(Registries.BIOME, PySift.id(biome));
        Pair<BlockPos, Holder<Biome>> r = level.findClosestBiome3d(h -> h.is(key), new BlockPos(0, 64, 0), 6400, 32, 8);
        return r == null ? null : r.getFirst();
    }

    private static int surface(ServerLevel level, int x, int z) {
        level.getChunk(x >> 4, z >> 4);
        return level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, x, z);
    }

    private void place(ClientGameTestContext context, TestServerContext server, String name, String cmd, boolean dark) {
        if (cmd == null) {
            System.out.println("WIKISHOT not found " + name);
            return;
        }
        server.runCommand(dark ? "effect give @p minecraft:night_vision infinite 0 true" : "effect clear @p");
        server.runCommand("execute in pysift:the_sift run time set 6000");
        server.runCommand(cmd);
        context.waitTicks(260);
        shot(context, name);
        System.out.println("WIKISHOT ok " + name + " " + cmd);
    }

    private void siftPlaces(ClientGameTestContext context, TestServerContext server, String only) {
        for (String b : only.isEmpty() || only.contains("biomes") ? SURFACE : new String[0]) {
            try {
                String cmd = server.computeOnServer(s -> {
                    BlockPos p = findBiome(s, b);
                    if (p == null) return null;
                    ServerLevel level = s.getLevel(sift());
                    int y = surface(level, p.getX(), p.getZ());
                    int ty = surface(level, p.getX(), p.getZ() + 40);
                    return tpCommand(p.getX(), Math.max(y, ty) + 22, p.getZ() - 10, p.getX(), ty + 2, p.getZ() + 40);
                });
                place(context, server, "biome_" + b, cmd, false);
            } catch (Throwable t) {
                System.out.println("WIKISHOT failed biome " + b + ": " + t);
            }
        }
        for (String b : only.isEmpty() || only.contains("biomes") ? UNDERGROUND : new String[0]) {
            try {
                String cmd = server.computeOnServer(s -> {
                    BlockPos p = findBiome(s, b);
                    if (p == null) return null;
                    ServerLevel level = s.getLevel(sift());
                    BlockPos best = null;
                    int bestScore = -1;
                    for (int dx = -16; dx <= 16; dx += 2) for (int dz = -16; dz <= 16; dz += 2) {
                        level.getChunk((p.getX() + dx) >> 4, (p.getZ() + dz) >> 4);
                        for (int dy = -12; dy <= 12; dy += 2) {
                            BlockPos q = p.offset(dx, dy, dz);
                            int score = 0;
                            for (int a = -3; a <= 3; a++) for (int c = -3; c <= 3; c++) for (int e = -2; e <= 2; e++) {
                                if (level.getBlockState(q.offset(a, e, c)).isAir()) score++;
                            }
                            if (score > bestScore && level.getBiome(q).is(ResourceKey.create(Registries.BIOME, PySift.id(b)))) {
                                bestScore = score;
                                best = q;
                            }
                        }
                    }
                    if (best == null) return null;
                    return tpCommand(best.getX() + 0.5, best.getY() + 1.5, best.getZ() + 0.5, best.getX() + 8, best.getY() - 2, best.getZ() + 14);
                });
                place(context, server, "biome_" + b, cmd, true);
            } catch (Throwable t) {
                System.out.println("WIKISHOT failed biome " + b + ": " + t);
            }
        }
        for (String st : only.isEmpty() || only.contains("structures") ? STRUCTURES : new String[0]) {
            try {
                String cmd = server.computeOnServer(s -> {
                    ServerLevel level = s.getLevel(sift());
                    Holder<Structure> h = s.registryAccess().lookupOrThrow(Registries.STRUCTURE)
                            .getOrThrow(ResourceKey.create(Registries.STRUCTURE, PySift.id(st)));
                    Pair<BlockPos, Holder<Structure>> r = level.getChunkSource().getGenerator()
                            .findNearestMapStructure(level, HolderSet.direct(h), BlockPos.ZERO, 120, false);
                    System.out.println("WIKISHOT debug " + st + " nearest=" + (r == null ? null : r.getFirst()));
                    if (r == null) return null;
                    BlockPos p = r.getFirst();
                    for (int dx = -4; dx <= 4; dx++) for (int dz = -4; dz <= 4; dz++) {
                        level.getChunk((p.getX() >> 4) + dx, (p.getZ() >> 4) + dz);
                    }
                    StructureStart start = level.getChunk(p.getX() >> 4, p.getZ() >> 4).getStartForStructure(h.value());
                    System.out.println("WIKISHOT debug " + st + " start=" + start + " valid=" + (start != null && start.isValid()));
                    if (start == null || !start.isValid()) {
                        start = level.structureManager().getStructureAt(p.atY(64), h.value());
                        if (start == null || !start.isValid()) {
                            for (int dx = -2; dx <= 2 && (start == null || !start.isValid()); dx++)
                                for (int dz = -2; dz <= 2 && (start == null || !start.isValid()); dz++)
                                    start = level.getChunk((p.getX() >> 4) + dx, (p.getZ() >> 4) + dz).getStartForStructure(h.value());
                        }
                        System.out.println("WIKISHOT debug " + st + " retry valid=" + (start != null && start.isValid()));
                        if (start == null || !start.isValid()) return null;
                    }
                    boolean den = st.equals("echo_den");
                    BoundingBox box = den ? start.getPieces().get(0).getBoundingBox() : start.getBoundingBox();
                    double cx = (box.minX() + box.maxX()) / 2.0 + 0.5, cz = (box.minZ() + box.maxZ()) / 2.0 + 0.5;
                    double cy = (box.minY() + box.maxY()) / 2.0;
                    if (den) {
                        return tpCommand(cx, box.minY() + 9, cz - box.getZSpan() * 0.42, cx, box.minY() + 2, cz + 2);
                    }
                    double size = Math.max(box.getXSpan(), box.getZSpan());
                    double dist = size * 1.05 + 6;
                    return tpCommand(cx + dist * 0.5, box.maxY() + size * 0.45 + 4, cz - dist * 0.866, cx, cy, cz);
                });
                place(context, server, "structure_" + st, cmd, st.equals("echo_den") || st.equals("dune_tomb"));
            } catch (Throwable t) {
                System.out.println("WIKISHOT failed structure " + st + ": " + t);
            }
        }
        for (String[] a : only.isEmpty() || only.contains("arenas") ? ARENAS : new String[0][]) {
            try {
                String cmd = server.computeOnServer(s -> {
                    BlockPos p = findBiome(s, a[1]);
                    if (p == null) return null;
                    ServerLevel level = s.getLevel(sift());
                    int x = p.getX() + 200, z = p.getZ();
                    BlockPos q = findBiome(s, a[1]);
                    for (int dx = -4; dx <= 4; dx++) for (int dz = -4; dz <= 4; dz++) {
                        level.getChunk((x >> 4) + dx, (z >> 4) + dz);
                    }
                    if (!level.getBiome(new BlockPos(x, 64, z)).is(ResourceKey.create(Registries.BIOME, PySift.id(a[1])))) {
                        x = q.getX();
                        z = q.getZ();
                    }
                    int y = surface(level, x, z);
                    BossArenaFeature.generateArenaDirectly(level, a[0], new BlockPos(x, y, z));
                    return tpCommand(x + 30, y + 34, z - 52, x, y + 2, z);
                });
                place(context, server, "arena_" + a[0], cmd, false);
            } catch (Throwable t) {
                System.out.println("WIKISHOT failed arena " + a[0] + ": " + t);
            }
        }
    }
